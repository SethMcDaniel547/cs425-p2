## Design
This design is split into three layers

### Packet Layer (top half of lab.c)
This block is responsible for cutting up the input (doesnt read the file thats I/O) to fit it into 10 byte packets and formats each packet for delivery. When packets arrive this is what also performs a checksum and rejects it if the checksum isnt correct

### Protocol Layer (bottom half of lab.c)
This is where we get that sender/receiver split. So the sender obviously sends the packets but it also keeps copies of the packets it hasnt gotten an acknowledge for yet (from that window) in case it needs to resend them. When it does get an acknowledge back, it moves over to the next window of packets to send. The receiver is the one that gets the files, but it only accepts them in the right order.

### I/O Layer (main.c)
This is the block of code that sets up the UDP sockets and parses the command line arguments. This is also where it reads and writes to files as mentioned in the packet layer section. 

### Why is it there?
Primary its there so its easier to test and ensure each piece works as expected, so when there is an issue its much easier to find (at least thats the most useful reason for me I found). One benefit of writing code in these more moduler pieces is that you can reuse code more efficiently.

uint16_t packet_checksum(const uint8_t *data, size_t length)
{
    uint32_t sum = 0;
    size_t i;
    for (i = 0; i + 1U < length; i += 2U) {
        sum += ((uint32_t)data[i] << 8U) | data[i + 1U];
        sum = (sum & 0xffffU) + (sum >> 16U);
    }
    if (i < length) {
        sum += (uint32_t)data[i] << 8U;
        sum = (sum & 0xffffU) + (sum >> 16U);
    }
    return (uint16_t)(~sum);
}

int packet_encode(const Packet *packet, uint8_t *wire, size_t capacity,
                  size_t *wire_length)
{
    size_t total;
    uint16_t checksum;
    if (packet == NULL || wire == NULL || wire_length == NULL ||
        packet->type > 2U || packet->reserved != 0U ||
        packet->length > PACKET_PAYLOAD_MAX ||
        ((packet->type == 1U || packet->type == 2U) && packet->length != 0U)) {
        return -1;
    }
    total = PACKET_HEADER_SIZE + packet->length;
    if (capacity < total) return -1;
    wire[0] = packet->type;
    wire[1] = 0;
    wire[2] = 0;
    wire[3] = 0;
    {
        uint32_t nseq = htonl(packet->seq);
        uint16_t nlen = htons(packet->length);
        memcpy(wire + 4, &nseq, sizeof nseq);
        memcpy(wire + 8, &nlen, sizeof nlen);
    }
    if (packet->length != 0U) memcpy(wire + PACKET_HEADER_SIZE, packet->payload, packet->length);
    checksum = packet_checksum(wire, total);
    {
        uint16_t nchecksum = htons(checksum);
        memcpy(wire + 2, &nchecksum, sizeof nchecksum);
    }
    *wire_length = total;
    return 0;
}

int packet_decode(const uint8_t *wire, size_t wire_length, Packet *packet)
{
    uint16_t received;
    uint16_t length;
    uint32_t seq;
    uint8_t copy[PACKET_MAX_SIZE];
    if (wire == NULL || packet == NULL || wire_length < PACKET_HEADER_SIZE ||
        wire_length > PACKET_MAX_SIZE) return -1;
    memcpy(&length, wire + 8, sizeof length);
    length = ntohs(length);
    if (length > PACKET_PAYLOAD_MAX || PACKET_HEADER_SIZE + length != wire_length ||
        wire[0] > 2U || wire[1] != 0U ||
        ((wire[0] == 1U || wire[0] == 2U) && length != 0U)) return -1;
    memcpy(copy, wire, wire_length);
    memcpy(&received, copy + 2, sizeof received);
    received = ntohs(received);
    copy[2] = 0;
    copy[3] = 0;
    if (packet_checksum(copy, wire_length) != received) return -1;
    packet->type = wire[0];
    packet->reserved = wire[1];
    packet->checksum = received;
    memcpy(&seq, wire + 4, sizeof seq);
    packet->seq = ntohl(seq);
    packet->length = length;
    if (length != 0U) memcpy(packet->payload, wire + PACKET_HEADER_SIZE, length);
    return 0;
}

static Packet packet_make(uint8_t type, uint32_t seq)
{
    Packet packet;
    memset(&packet, 0, sizeof packet);
    packet.type = type;
    packet.seq = seq;
    return packet;
}

Packet packet_make_data(uint32_t seq, const uint8_t *data, size_t length)
{
    Packet packet = packet_make(0, seq);
    if (length > PACKET_PAYLOAD_MAX) {
        /*
         * The API does not receive the source buffer's capacity, so an
         * oversized request cannot be safely truncated with memcpy.
         * Preserve the packet-size contract and leave the payload zeroed.
         */
        packet.length = PACKET_PAYLOAD_MAX;
        return packet;
    }
    packet.length = (uint16_t)length;
    if (length != 0U && data != NULL) memcpy(packet.payload, data, length);
    return packet;
}

ReceiverEvent receiver_packet(ReceiverState *state, const Packet *packet)
{
    ReceiverEvent event;
    memset(&event, 0, sizeof event);
    if (state == NULL || packet == NULL) return event;
    event.accepted = 1;
    event.ack = packet_make_ack(state->expected);
    if (packet->type == 0U && !state->finished && packet->seq == state->expected) {
        state->expected++;
        event.deliver = 1;
        event.payload = packet->payload;
        event.payload_length = packet->length;
        event.ack = packet_make_ack(state->expected);
    } else if (packet->type == 2U && !state->finished && packet->seq == state->expected &&
               packet->length == 0U) {
        state->expected++;
        state->finished = 1;
        event.finished = 1;
        event.ack = packet_make_ack(state->expected);
    } else if (packet->type != 0U && packet->type != 2U) {
        event.accepted = 0;
    }

}

int sender_init(SenderState *state, const uint8_t *data, size_t length,
                uint32_t window, uint64_t now_ms, uint64_t timeout_ms)
{
    size_t count = (length + PACKET_PAYLOAD_MAX - 1U) / PACKET_PAYLOAD_MAX;
    size_t i;
    if (state == NULL) return -1;
    memset(state, 0, sizeof *state);
    state->packet_count = count;
    state->window = window < 1U ? 1U : (window > SENDER_MAX_WINDOW ? SENDER_MAX_WINDOW : window);
    state->packets = calloc(count + 1U, sizeof *state->packets);
    if (state->packets == NULL) return -1;
    for (i = 0; i < count; i++) {
        size_t offset = i * PACKET_PAYLOAD_MAX;
        size_t remaining = length - offset;
        state->packets[i] = packet_make_data((uint32_t)i, data + offset,
                                              remaining > PACKET_PAYLOAD_MAX ? PACKET_PAYLOAD_MAX : remaining);
    }
    state->packets[count] = packet_make_fin((uint32_t)count);
    state->deadline_ms = now_ms + timeout_ms;
    return 0;
}

int sender_next_packet(SenderState *state, Packet *packet, uint64_t now_ms,
                       uint64_t timeout_ms)
{
    uint32_t total;
    if (state == NULL || packet == NULL || state->packets == NULL ||
        state->complete) return 0;
    total = (uint32_t)state->packet_count + 1U;
    if (state->next >= total || state->next >= state->base + state->window) return 0;
    *packet = state->packets[state->next++];
    if (!state->timer_running) {
        state->timer_running = 1;
        state->deadline_ms = now_ms + timeout_ms;
    }
    return 1;
}

int sender_ack(SenderState *state, uint32_t ack_seq, uint64_t now_ms,
               uint64_t timeout_ms)
{
    uint32_t total;
    if (state == NULL) return 0;
    total = (uint32_t)state->packet_count + 1U;
    if (state == NULL || ack_seq <= state->base || ack_seq > total || ack_seq > state->next) return 0;
    state->base = ack_seq;
    state->timeout_count = 0;
    if (state->base == total) {
        state->complete = 1;
        state->timer_running = 0;
    } else {
        state->timer_running = state->base < state->next;
        if (state->timer_running) state->deadline_ms = now_ms + timeout_ms;
    }
    return 1;
}

int sender_timeout(SenderState *state, uint64_t now_ms, uint64_t timeout_ms)
{
    if (state == NULL || !state->timer_running || now_ms < state->deadline_ms) return 0;
    state->timeout_count++;
    if (state->timeout_count >= 10U) return -1;
    state->deadline_ms = now_ms + timeout_ms;
    return 1;
}

int sender_done(const SenderState *state) { return state != NULL && state->complete; }

        action.deadline_ms = state->deadline_ms;
        action.timer_running = state->timer_running;
        action.complete = state->complete;
    }
    return action;
}

SenderAction sender_send_ready(SenderState *state, uint64_t now_ms,
                               uint64_t timeout_ms)
{
    SenderAction action = sender_action(state);
    Packet packet;
    if (state == NULL) {
        action.failed = 1;
        return action;
    }
    while (action.packet_count < SENDER_MAX_WINDOW &&
           sender_next_packet(state, &packet, now_ms, timeout_ms) != 0) {
        action.packets[action.packet_count++] = packet;
    }
    action.deadline_ms = state->deadline_ms;
    action.timer_running = state->timer_running;
    action.complete = state->complete;
    return action;
}

SenderAction sender_acknowledged(SenderState *state, uint32_t ack_seq,
                                 uint64_t now_ms, uint64_t timeout_ms)
{
    SenderAction action;
    if (state == NULL || sender_ack(state, ack_seq, now_ms, timeout_ms) == 0) {
        action = sender_action(state);
        return action;
    }
    return sender_send_ready(state, now_ms, timeout_ms);
}

SenderAction sender_timer_expired(SenderState *state, uint64_t now_ms,
                                  uint64_t timeout_ms)
{
    SenderAction action = sender_action(state);
    uint32_t sequence;
    if (state == NULL) {
        action.failed = 1;
        return action;
    }
    if (sender_timeout(state, now_ms, timeout_ms) < 0) {
        action.failed = 1;
        return action;
    }
    if (state->timer_running) {
        for (sequence = state->base;
             sequence < state->next && action.packet_count < SENDER_MAX_WINDOW;
             sequence++) {
            action.packets[action.packet_count++] = state->packets[sequence];
        }
    }
    action.deadline_ms = state->deadline_ms;
    action.timer_running = state->timer_running;
    action.complete = state->complete;
    return action;
}

static uint64_t monotonic_ms(void)
{
    struct timespec ts;
    (void)clock_gettime(CLOCK_MONOTONIC, &ts);
    return (uint64_t)ts.tv_sec * 1000U + (uint64_t)ts.tv_nsec / 1000000U;
}

static int resolve_relay(const char *host, const char *port, struct sockaddr_storage *address,
                         socklen_t *address_length)
{
    struct addrinfo hints;
    struct addrinfo *result;
    memset(&hints, 0, sizeof hints);
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_DGRAM;
    if (getaddrinfo(host, port, &hints, &result) != 0) return -1;
    memcpy(address, result->ai_addr, result->ai_addrlen);
    *address_length = (socklen_t)result->ai_addrlen;
    freeaddrinfo(result);
    return 0;
}

static int valid_session(const char *session)
{
    size_t i;
    for (i = 0; session[i] != '\0'; i++) {
        if (!((session[i] >= 'a' && session[i] <= 'z') ||
              (session[i] >= '0' && session[i] <= '9') || session[i] == '-')) return 0;
    }
    return i >= 1U && i <= 32U;
}

static int hello(int socket_fd, const struct sockaddr *relay, socklen_t relay_length,
                 const char *message)
{
    char reply[256];
    ssize_t received;
    struct pollfd descriptor = {socket_fd, POLLIN, 0};
    int attempt;
    for (attempt = 0; attempt < 5; attempt++) {
        if (sendto(socket_fd, message, strlen(message), 0, relay, relay_length) < 0) return -1;
        if (poll(&descriptor, 1, 1000) <= 0) continue;
        received = recvfrom(socket_fd, reply, sizeof reply - 1U, 0, NULL, NULL);
        if (received < 0) return -1;
        reply[received] = '\0';
        if (strncmp(reply, "OK", 2) == 0) return 0;
        fprintf(stderr, "%s\n", reply);
        return -2;
    }
    fprintf(stderr, "registration timed out\n");
    return -1;
}

static int send_packet(int fd, const struct sockaddr *relay, socklen_t relay_length,
                       const Packet *packet)
{
    uint8_t wire[PACKET_MAX_SIZE];
    size_t length;
    if (packet_encode(packet, wire, sizeof wire, &length) != 0) return -1;
    {
        ssize_t sent = sendto(fd, wire, length, 0, relay, relay_length);
        if (sent < 0 || (size_t)sent != length) {
            perror("sendto");
            return -1;
        }
    }
    return 0;
}

static int receive_packet(int fd, Packet *packet)
{
    uint8_t wire[PACKET_MAX_SIZE];
    ssize_t length = recvfrom(fd, wire, sizeof wire, 0, NULL, NULL);
    if (length < 0) return -1;
    return packet_decode(wire, (size_t)length, packet);
}

static int send_action(int fd, const struct sockaddr *relay, socklen_t relay_length,
                       const SenderAction *action)
{
    size_t i;
    for (i = 0; i < action->packet_count; i++) {
        if (send_packet(fd, relay, relay_length, &action->packets[i]) != 0) return -1;
    }
    return 0;
}

static int run_receiver(int fd, const struct sockaddr *relay, socklen_t relay_length,
                        const char *output_path)
{
    FILE *output = fopen(output_path, "wb");
    ReceiverState state;
    uint64_t last = monotonic_ms();
    if (output == NULL) return 2;
    receiver_init(&state);
    while (monotonic_ms() - last < 30000U) {
        struct pollfd descriptor = {fd, POLLIN, 0};
        Packet packet;
        int ready = poll(&descriptor, 1, 1000);
        if (ready <= 0) continue;
        if (receive_packet(fd, &packet) != 0) continue;
        last = monotonic_ms();
        {
            ReceiverEvent event = receiver_packet(&state, &packet);
            if (!event.accepted) continue;
            if (event.deliver && fwrite(event.payload, 1, event.payload_length, output) != event.payload_length) {
                fclose(output);
                return 2;
            }
            if (send_packet(fd, relay, relay_length, &event.ack) != 0) {
                fclose(output);
                return 2;
            }
            if (event.finished) {
                uint64_t linger_end = monotonic_ms() + 2000U;
                (void)fclose(output);
                while (monotonic_ms() < linger_end) {
                    struct pollfd linger = {fd, POLLIN, 0};
                    if (poll(&linger, 1, 100) > 0 && receive_packet(fd, &packet) == 0) {
                        ReceiverEvent repeated = receiver_packet(&state, &packet);
                        if (repeated.accepted &&
                            send_packet(fd, relay, relay_length, &repeated.ack) != 0) {
                            return 2;
                        }
                    }
                }
                return 0;
            }
        }
    }
    (void)fclose(output);
    return 2;
}

static int run_sender(int fd, const struct sockaddr *relay, socklen_t relay_length,
                      const char *input_path, uint32_t window, uint64_t timeout_ms)
{
    FILE *input = fopen(input_path, "rb");
    long file_size;
    uint8_t *data;
    SenderState state;
    if (input == NULL || fseek(input, 0, SEEK_END) != 0) {
        fprintf(stderr, "cannot open input file %s\n", input_path);
        if (input != NULL) fclose(input);
        return 2;
    }
    file_size = ftell(input);
    if (file_size < 0 || fseek(input, 0, SEEK_SET) != 0) { fclose(input); return 2; }
    data = malloc((size_t)file_size == 0U ? 1U : (size_t)file_size);
    if (data == NULL || ((size_t)file_size != 0U && fread(data, 1, (size_t)file_size, input) != (size_t)file_size)) {
        free(data); fclose(input); return 2;
    }
    fclose(input);
    if (sender_init(&state, data, (size_t)file_size, window, monotonic_ms(), timeout_ms) != 0) {
        free(data);
        return 2;
    }
    free(data);
    if (!sender_ready(&state)) return 2;
    while (!sender_done(&state)) {
        uint64_t now = monotonic_ms();
        Packet packet;
        SenderAction action = sender_send_ready(&state, now, timeout_ms);
        if (send_action(fd, relay, relay_length, &action) != 0) {
            sender_destroy(&state);
            return 2;
        }
        {
            int wait_ms = action.timer_running
                ? (action.deadline_ms > now ? (int)(action.deadline_ms - now) : 0)
                : 1000;
            struct pollfd descriptor = {fd, POLLIN, 0};
            int ready = poll(&descriptor, 1, wait_ms);
            if (ready > 0) {
                if (receive_packet(fd, &packet) == 0 && packet.type == 1U &&
                    packet.length == 0U) {
                    action = sender_acknowledged(&state, packet.seq, monotonic_ms(),
                                                 timeout_ms);
                    if (action.failed || send_action(fd, relay, relay_length, &action) != 0) {
                        sender_destroy(&state);
                        return 2;
                    }
                }
            } else if (ready < 0 && errno != EINTR) {
                sender_destroy(&state);
                return 2;
            }
            if (action.timer_running && monotonic_ms() >= action.deadline_ms) {
                action = sender_timer_expired(&state, monotonic_ms(), timeout_ms);
                if (action.failed) {
                    fprintf(stderr, "giving up after 10 timeouts\n");
                    sender_destroy(&state);
                    return 2;
                }
                if (send_action(fd, relay, relay_length, &action) != 0) {
                    sender_destroy(&state);
                    return 2;
                }
            }
        }
    }
    sender_destroy(&state);
    return 0;
}

int main(int argc, char **argv)
{
    const char *mode;
    const char *session = NULL;
    const char *relay_host;
    const char *file_path;
    char port_text[16];
    int port = 4250;
    int window = 8;
    int timeout = 250;
    double loss = 0.0;
    double corrupt = 0.0;
    double dup = 0.0;
    int option;
    int fd;
    struct sockaddr_storage relay;
    socklen_t relay_length;
    if (argc == 1) { usage(); return 0; }
    if (argc < 2 || (strcmp(argv[1], "send") != 0 && strcmp(argv[1], "recv") != 0)) { usage(); return 1; }
    mode = argv[1];
    optind = 2;
    while ((option = getopt(argc, argv, "s:w:T:l:c:d:p:")) != -1) {
        char *end;
        switch (option) {
        case 's': session = optarg; break;
        case 'w': window = (int)strtol(optarg, &end, 10); if (*end != '\0') return 1; break;
        case 'T': timeout = (int)strtol(optarg, &end, 10); if (*end != '\0') return 1; break;
        case 'l': loss = strtod(optarg, &end); if (*end != '\0') return 1; break;
        case 'c': corrupt = strtod(optarg, &end); if (*end != '\0') return 1; break;
        case 'd': dup = strtod(optarg, &end); if (*end != '\0') return 1; break;
        case 'p': port = (int)strtol(optarg, &end, 10); if (*end != '\0') return 1; break;
        default: usage(); return 1;
        }
    }
    if (session == NULL || !valid_session(session) || optind + 2 != argc ||
        port < 1 || port > 65535 || timeout <= 0 || window < 1 || window > 64 ||
        loss < 0.0 || loss > 0.5 || corrupt < 0.0 || corrupt > 0.5 || dup < 0.0 || dup > 0.5) return 1;
    relay_host = argv[optind];
    file_path = argv[optind + 1];
    (void)snprintf(port_text, sizeof port_text, "%d", port);
    if (resolve_relay(relay_host, port_text, &relay, &relay_length) != 0) return 2;
    fd = socket(relay.ss_family, SOCK_DGRAM, 0);
    if (fd < 0) return 2;
    {
        char message[256];
        int hello_result;
        if (strcmp(mode, "recv") == 0) {
            (void)snprintf(message, sizeof message, "HELLO %s recv", session);
        } else {
            (void)snprintf(message, sizeof message, "HELLO %s send %.17g %.17g %.17g",
                           session, loss, corrupt, dup);
        }
        hello_result = hello(fd, (const struct sockaddr *)&relay, relay_length, message);
        if (hello_result != 0) { close(fd); return 2; }
    }
    {
        int result = strcmp(mode, "recv") == 0
            ? run_receiver(fd, (const struct sockaddr *)&relay, relay_length, file_path)
            : run_sender(fd, (const struct sockaddr *)&relay, relay_length, file_path,
                         (uint32_t)window, (uint64_t)timeout);
        close(fd);
        return result;
    }
}

static void queue_packet(Packet *queue, size_t *count, const Packet *packet,

void test_in_memory_transfer_with_loss_corruption_and_duplicates(void) {
  uint8_t input[2500];
  uint8_t output[2500];
  static Packet data_queue[4096];
  static Packet ack_queue[4096];
  unsigned seeds[] = {7U, 19U, 41U};
  size_t seed_index;

  for (size_t i = 0; i < sizeof input; i++) input[i] = (uint8_t)(i * 37U);
  for (seed_index = 0; seed_index < sizeof seeds / sizeof seeds[0]; seed_index++) {
    SenderState sender;
    ReceiverState receiver;
    SenderAction action;
    unsigned seed = seeds[seed_index];
    size_t data_count = 0;
    size_t ack_count = 0;
    size_t delivered = 0;
    uint64_t now = 0;
    unsigned steps;

    TEST_ASSERT_EQUAL_INT(0, sender_init(&sender, input, sizeof input, 4, now, 5));
    receiver_init(&receiver);
    action = sender_send_ready(&sender, now, 5);
    queue_action(data_queue, &data_count, &action, &seed);

    for (steps = 0; steps < 10000U && !sender_done(&sender); steps++) {
      if (data_count != 0U) {
        Packet packet = data_queue[--data_count];
        ReceiverEvent event = receiver_packet(&receiver, &packet);
        if (event.deliver) {
          TEST_ASSERT_TRUE(delivered + event.payload_length <= sizeof output);
          memcpy(output + delivered, event.payload, event.payload_length);
          delivered += event.payload_length;
        }
        if (event.accepted) {
          queue_packet(ack_queue, &ack_count, &event.ack, &seed);
        }
      } else if (ack_count != 0U) {
        Packet ack = ack_queue[--ack_count];
        action = sender_acknowledged(&sender, ack.seq, now, 5);
        TEST_ASSERT_FALSE(action.failed);
        queue_action(data_queue, &data_count, &action, &seed);
      } else {
        now += 5;
        action = sender_timer_expired(&sender, now, 5);
        TEST_ASSERT_FALSE(action.failed);
        queue_action(data_queue, &data_count, &action, &seed);
      }
    }
    TEST_ASSERT_TRUE(sender_done(&sender));
    TEST_ASSERT_EQUAL_UINT32(sizeof input, delivered);
    TEST_ASSERT_EQUAL_MEMORY(input, output, sizeof input);
    sender_destroy(&sender);
  }
}
